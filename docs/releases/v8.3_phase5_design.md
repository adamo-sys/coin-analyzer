## Scope

**Phase 5 adds a GUI dialog for the Collector Workspace.** It displays all 10 panels and the Reports Panel using the existing `ttk.Notebook` + `tk.Text` patterns. It is read-only — all mutation flows through "Open in Tool..." buttons that launch existing dialogs.

- `open_collector_workspace()` method in `coin_collection_gui.py`
- `ttk.Notebook` with 11 tabs (10 panels + Reports)
- Each tab displays read-only text from workspace getters
- "Refresh Workspace" button
- "Open in [Tool]..." buttons per tab for mutation actions
- Export buttons where applicable (delegate to workspace `export_report()`)
- Menu item under "Collector Home" → "Collector Workspace"
- Unit tests (import checks, smoke tests)

### Out of Scope (confirmed)
- No new business logic in the GUI
- No new panels beyond the 10 already implemented
- No new report types beyond the 16 already registered
- No collection mutation in the workspace GUI
- No persistence changes
- No engine changes

```python
def open_collector_workspace(self):
    """Open Collector Workspace dialog with notebook tabs for all panels.

    Read-only display. All mutation flows through 'Open in Tool...' buttons.
    """
    # Create workspace instance with current context
    workspace = CollectorWorkspace(
        self._collection_items(),
        want_list_intents=self._active_want_list_intents(),
        photo_records=self.photo_records,
        shopping_candidates=self.shopping_candidates,
        market_awareness_engine=self.market_awareness_engine,
        photo_candidates=getattr(self, "photo_candidates", None),
        watchlists=getattr(self, "watchlists", None),
        ocr_reports=getattr(self, "ocr_reports", None),
        workflow_statuses=getattr(self, "workflow_statuses", None),
        acknowledged_action_ids=getattr(self, "acknowledged_action_ids", None),
    )

    dialog = tk.Toplevel(self.root)
    dialog.title("Collector Workspace")
    dialog.geometry("1000x800")

    main_frame = ttk.Frame(dialog, padding="10")
    main_frame.pack(fill=tk.BOTH, expand=True)

    # Notebook
    notebook = ttk.Notebook(main_frame)
    notebook.pack(fill=tk.BOTH, expand=True)

    # Create tabs
    tabs = self._create_workspace_tabs(notebook, workspace)

    # Button bar
    button_frame = ttk.Frame(main_frame)
    button_frame.pack(fill=tk.X, pady=(10, 0))

    def refresh_workspace():
        workspace.refresh()
        self._refresh_workspace_tabs(tabs, workspace)

    ttk.Button(button_frame, text="Refresh Workspace", command=refresh_workspace).pack(side=tk.LEFT, padx=(0, 6))
    ttk.Button(button_frame, text="Close", command=dialog.destroy).pack(side=tk.LEFT)

def _format_dashboard(self, report: DashboardReport) -> str:
    """Format dashboard report as readable text."""
    lines = ["# Dashboard", ""]
    lines.append(f"Health Score: {report.health_score or 'N/A'}")
    lines.append(f"Quality Score: {report.quality_score or 'N/A'}")
    lines.append(f"Integrity Score: {report.integrity_score or 'N/A'}")
    lines.append(f"Top Priority: {report.top_priority or 'None'}")
    lines.append(f"Best Next Purchase: {report.best_next_purchase or 'None'}")
    lines.append(f"Data Safety: {report.data_safety_status or 'Unknown'}")
    lines.append(f"Backup Ready: {'Yes' if report.backup_ready else 'No'}")
    if report.todays_tasks:
        lines.extend(["", "## Today's Tasks"])
        for task in report.todays_tasks:
            lines.append(f"- {task}")
    if report.recent_activity:
        lines.extend(["", "## Recent Activity"])
        for activity in report.recent_activity:
            lines.append(f"- {activity}")
    if report.engine_errors:
        lines.extend(["", "## ⚠️ Warnings"])
        for error in report.engine_errors:
            lines.append(f"- {error}")
    return "\n".join(lines) + "\n"
```

## Report Display/Export Flow

The Reports tab uses `get_reports()` to display the menu:

```python
def _create_reports_tab(self, notebook, workspace):
    frame = ttk.Frame(notebook, padding="10")
    notebook.add(frame, text="📈 Reports")

    # Treeview or list of reports
    menu = workspace.get_reports()
    for category in menu.categories:
        ttk.Label(frame, text=category, font=("TkDefaultFont", 10, "bold")).pack(anchor=tk.W, pady=(10, 0))
        for report in menu.by_category(category):
            row = ttk.Frame(frame)
            row.pack(fill=tk.X, pady=2)
            status = "✓" if report.available else "✗"
            ttk.Label(row, text=f"  {status} {report.title}").pack(side=tk.LEFT)
            if report.available:
                ttk.Button(row, text="Generate", command=lambda r=report: self._generate_report(workspace, r)).pack(side=tk.RIGHT, padx=(6, 0))
                if report.has_markdown_export:
                    ttk.Button(row, text="MD", command=lambda r=report: self._export_report(workspace, r, "markdown")).pack(side=tk.RIGHT, padx=(6, 0))
                if report.has_csv_export:
                    ttk.Button(row, text="CSV", command=lambda r=report: self._export_report(workspace, r, "csv")).pack(side=tk.RIGHT, padx=(6, 0))
```

## Implementation Details

### Formatting helpers

Each panel needs a `_format_*()` method that converts the report dataclass into readable text. These are pure formatting — no business logic.

```python
def _format_inbox(self, report: InboxReport) -> str:
    lines = ["# Inbox", ""]
    lines.append(f"Total Pending: {report.total_pending}")
    lines.append(f"Collection Assistant: {report.collection_assistant_pending}")
    lines.append(f"Batch Processing: {report.batch_processing_pending}")
    lines.append(f"AI Grading: {report.ai_grading_review}")
    if report.items:
        lines.extend(["", "## Items"])
        for item in report.items:
            lines.append(f"- [{item['source']}] {item['label']} (confidence: {item['confidence']:.0%})")
    if report.engine_errors:
        lines.extend(["", "## ⚠️ Warnings"])
        for error in report.engine_errors:
            lines.append(f"- {error}")
    return "\n".join(lines) + "\n"
```

### Refresh mechanism

When "Refresh Workspace" is clicked:
1. `workspace.refresh()` clears cache
2. `_refresh_workspace_tabs()` re-calls each `get_*()` and updates the text widget
3. Text widgets are temporarily enabled, cleared, re-populated, then disabled

```python
def _refresh_workspace_tabs(self, tabs, workspace):

    text.delete("1.0", tk.END)
```

## Guardrails

- **Read-only display:** No collection mutation in the workspace GUI. All mutation through "Open in Tool..." buttons.
- **No business logic:** Formatting only. The workspace does the aggregation.
- **No new panels:** Only display the 10 panels already implemented.
- **No new reports:** Only display the 16 reports already registered.
- **Delegated export:** Export buttons call `workspace.export_report()` which delegates to engines.
- **Error resilience:** Engine errors displayed as warnings, not crashes.
- **Lazy loading:** The workspace is created when the dialog opens, not at startup.
